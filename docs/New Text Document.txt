1. Baseline stream payload
When a single-agent turn produces a plain assistant message, the controller emits a cohesive_result event. The payload bundles narrative text, chart spec, SQL outputs, accessory context, and manifest metadata, while apply_mode_metadata and the instrumentation layer add mode, scheduling badges, and sequencing fields (seq/data.sequence, parallel_group, schedule_stage). A follow-up route tag is appended so the UI can keep routing hints.

{
  "event": "cohesive_result",
  "seq": 27,
  "data": {
    "analysis": "…",
    "analysis_length": 542,
    "chart_spec": { "mark": "line", "encoding": { "x": "...", "y": "..." } },
    "sql": "SELECT …",
    "sql_row_count": 128,
    "columns": ["quarter", "revenue"],
    "data_sample": [{ "quarter": "2024-Q2", "revenue": 8.1e9 }],
    "stock_widget": { "ticker": "NVDA", "price": 102.3 },
    "web_context": { "summary": "…", "snippets": ["…"] },
    "tool_manifest": [{ "name": "sql_generator", "description": "…" }],
    "tool_results": [{ "tool": "analysis_writer", "status": "completed" }],
    "analysis_sources": { "sql": true, "stock": true, "web": true },
    "mode": "single-agent",
    "badges": { "mode": "deterministic" },
    "accessory_strategy": "hedged",
    "parallel_group": "analysis",
    "schedule_stage": "analysis",
    "sequence": 27,
    "follow_up_route": "full_pipeline",
    "ts": "2025-11-04T12:00:00Z"
  }
}
2. Tool invocation delta
Tool turns surface as tool_call deltas. The “start” chunk captures which planner step is underway, while the matching “end” chunk includes elapsed time and tool-specific details (e.g., SQL template ID, generated SQL length). Mode metadata, routing hints, and sequencing fields are injected the same way as above.

{
  "event": "tool_call",
  "seq": 12,
  "data": {
    "tool": "sql_generator",
    "status": "end",
    "step": "sql_generated",
    "ts": "2025-11-04T11:59:12Z",
    "elapsed_ms": 1843,
    "details": {
      "llm_used": "gpt-4.1",
      "template_used": "peer_comparison_v2",
      "sql_length": 612
    },
    "latency_budget_ms": 2000,
    "output_artifacts": ["sql", "columns"],
    "concurrency_limit": 1,
    "mode": "single-agent",
    "badges": { "mode": "deterministic" },
    "parallel_group": "sql",
    "schedule_stage": "sql_generation",
    "sequence": 12,
    "follow_up_route": "full_pipeline"
  }
}
3. Final result delivery
After the agent run finishes, the controller emits an analysis_bundle event that wraps the cohesive payload alongside session metadata (session_id, flow, query), followed by a workflow_complete summary that reports the total elapsed time. The run itself is configured with a UUID trace_id, making it easy to propagate once exposed in downstream payloads.

{
  "event": "analysis_bundle",
  "seq": 38,
  "data": {
    "analysis_bundle": { "analysis": "…", "chart_spec": { … }, "tool_results": [ … ] },
    "session_id": "sess_123",
    "flow": "single-agent",
    "query": "How are Nvidia's margins trending?",
    "mode": "single-agent",
    "badges": { "mode": "deterministic" },
    "parallel_group": "analysis",
    "schedule_stage": "analysis",
    "sequence": 38,
    "ts": "2025-11-04T12:00:45Z"
  }
}
{
  "event": "workflow_complete",
  "seq": 39,
  "data": {
    "total_elapsed_ms": 48231,
    "mode": "single-agent",
    "badges": { "mode": "deterministic" },
    "parallel_group": "workflow",
    "schedule_stage": "workflow_complete",
    "sequence": 39,
    "ts": "2025-11-04T12:00:45Z"
  }
}
